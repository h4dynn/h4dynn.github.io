12 * Error handling helpers *

14/// Implementation for throwing an error with a trace (`ErrorTrace`) or without a trace (`Error`)
15pub trait ReturnError
16{
17  /// The ! return type shows that `fatal` should never end up returning
18  fn fatal(self) -> !;
19  /// For non-fatal errors only
20  fn warn(self);
21}
22
23/// Fatal, non-recoverable error handling - exit with message
24pub trait HandleError: Sized
25{
26  /// [`Result::Ok`]
27  type OkType;
28  /// [`Result::Err`]
29  type ErrorType: ReturnError;
30
31  /// This functions like `unwrap`: Return contents if OK or handle error (`Error::fatal`) if Err
32  fn handle(self) -> Self::OkType;
33  /// Warn on error, if an error occurs `None` is returned, if not then `Some` instead
34  fn or_warn(self) -> Option<Self::OkType>;
35}
36
37/// Inspect error & throw only a warning upon non-fatal invariants or fatal error handling
38pub trait AssessError<OkType, ErrType>
39{
40  /// Do not return an error if `optional` is true
41  /**
42    * # Errors
43    *
44    * * If non-optional and of error kind, carry the error over to output
45    */
46  fn assess(self, optional: bool) -> Result<Option<OkType>, ErrType>;
47}
48
49/**
50  * Addon more subject to an error for diagnostics, this is usually the name of something,
51  * i.e., if an error occurs with a specific service we add the service's name as subject
52  */
53pub trait ExtendWithSubject<OkType, ErrType>
54{
55  /**
56    * Add subject to an existing trace error
57    *
58    * # Errors
59    *
60    * * Result is of error variant
61    */
62  fn subject(self, subject: impl Display) -> Result<OkType, ErrType>;
63}
64
65/**
66  * Convert a non-standard error input into an `ErrorTrace`. This is usually done on a
67  * `Result<T, E>` but can also be done on a `Option<E>` where we assume the option's
68  * `Some` variant contains the error
69  */
70pub trait ErrorResult<ErrKind: ReturnError, ErrOutput: ReturnError, OkType, ErrType: Display>
71{
72  /// Convert an error to a trace without subject
73  /**
74    * # Errors
75    *
76    * * Data type contains an error (e.g. Result is Err(..))
77    */
78  fn into_trace(self, kind: ErrKind) -> Result<OkType, ErrOutput>;
79}

85/// Colourize an input ASCII type
86pub trait Colourize: Display + Sized
87{
88  /// Colour a string-like type
89  fn colour(self, colour: Colour) -> FmtString
90  {
91    FmtString { colours: vec![colour], inner: self.to_string(), dont_reset: false }
92  }
93
94  /// Alias to `self.colour(Colour::Bold)`
95  fn bold(self) -> FmtString
96  {
97    self.colour(Colour::Bold)
98  }
99}
100
101/// Colours & stylizing options (reset/bold)
102#[derive(PartialEq, Eq, Copy, Clone, Debug, Default, derive_more::Display)]
103pub enum Colour
104{
105  /// Clear previous formatting, reset to console default text
106  #[default]
107  #[display("\x1b[0m")]
108  Reset,
109  /// Make text bold, compatible with other colour provided before/after
110  #[display("\x1b[1m")]
111  Bold,
112  #[display("\x1b[1;31m")]
113  Red,
114  #[display("\x1b[1;33m")]
115  Orange,
116  #[display("\x1b[1;92m")]
117  Green
118}

120/// A borrowed version of the [`FmtString`], like how [`&str`] is to [`String`]
121#[derive(PartialEq, Eq, Copy, Clone, Debug, Default)]
122pub struct FmtStr<'inner>
123{
124  // Borrowed slice of inner values, not ideal to tie a lifetime to this struct but we have to
125  colours: &'inner [Colour],
126  inner: &'inner str,
127  // Prevent console reset (0x1b5b306d, "\x1b[0m") from being appended at the end of formatting
128  dont_reset: bool
129}

131/// An owned form of a colour formatted string (borrowed form: [`FmtStr`])
132#[derive(PartialEq, Eq, Clone, Debug, Default)]
133pub struct FmtString
134{
135  colours: Vec<Colour>,
136  inner: String,
137  dont_reset: bool
138}
139
140impl Display for FmtStr<'_>
141{
142  fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result
143  {
144    write!(f, "{}", self.format())
145  }
146}

148impl Display for FmtString
149{
150  fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result
151  {
152    // Reference to borrowed form first
153    write!(f, "{}", self.as_ref().format())
154  }
155}

157impl FmtString
158{
159  #[must_use]
160  pub const fn new() -> Self
161  {
162    Self { colours: Vec::new(), inner: String::new(), dont_reset: false }
163  }
164
165  /// Return `self` with additional `colour`. See also [`push_colour`]
166  #[must_use]
167  pub fn colour(mut self, colour: Colour) -> Self
168  {
169    self.push_colour(colour);
170    self
171  }
172
173  #[must_use]
174  pub fn bold(self) -> Self
175  {
176    self.colour(Colour::Bold)
177  }
178
179  /// Create a reference to this instance
180  #[must_use]
181  pub const fn as_ref(&self) -> FmtStr<'_>
182  {
183    FmtStr { colours: self.colours.as_slice(), inner: self.inner.as_str(), dont_reset: self.dont_reset }
184  }
185
186  // Change whether the end of the formatted String will have a reset (\e[0m)
187  #[must_use]
188  pub fn reset(self, reset: bool) -> Self
189  {
190    FmtString { colours: self.colours, inner: self.inner, dont_reset: !reset }
191  }
192
193  pub fn push_str(&mut self, string: &str)
194  {
195    self.inner.push_str(string);
196  }
197
198  pub fn push_colour(&mut self, colour: Colour)
199  {
200    self.colours.push(colour);
201  }
202
203  /// Clear the internal string contents
204  /**
205    * # Note
206    *
207    * This does not clear the colours. Refer to [`make_plain`]
208    */
209  pub fn clear(&mut self)
210  {
211    self.inner.clear();
212  }
213
214  /// Clear all colours from this instance
215  pub fn make_plain(&mut self)
216  {
217    self.colours = Vec::new();
218  }
219}
220
221impl FmtStr<'static>
222{
223  #[must_use]
224  pub const fn new() -> Self
225  {
226    Self { colours: &[], inner: "", dont_reset: false }
227  }
228}

230impl FmtStr<'_>
231{
232  /// Convert from borrowed form to owned instance
233  #[must_use]
234  pub fn to_owned(self) -> FmtString
235  {
236    FmtString { colours: self.colours.to_vec(), inner: self.inner.to_owned(), dont_reset: self.dont_reset }
237  }
238
239  /// Format to common string type
240  pub fn format(&self) -> String
241  {
242    // Transform all colours into their string data
243    let mut out: Vec<String> = self.colours.iter().map(ToString::to_string).collect();
244    // The main actual content
245    out.push(self.inner.to_string());
246
247    // This is going to be false 99% of the time except for some niche cases (change with `.reset(..)`)
248    if (!self.dont_reset)
249    {
250      out.push(Colour::Reset.to_string());
251    }
252
253    // Concatenate all output together into a singular string
254    out.join("")
255  }
256}

297/*
298 * What to do when a Error is found
299 * Implemented for Result<anything, error> and Option<error>
300 */
301impl<S, F: ReturnError> HandleError for Result<S, F>
302{
303  type OkType = S;
304  type ErrorType = F;
305
306  fn handle(self) -> Self::OkType
307  {
308    match (self)
309    {
310      Ok(ok) => ok,
311      // Handle error invariant with exit method, so OK type will always be returned in code order
312      Err(error) => error.fatal()
313    }
314  }
315  fn or_warn(self) -> Option<Self::OkType>
316  {
317    // Throw warning for error invariant & then extract Ok variant
318    self.map_err(ReturnError::warn).ok()
319  }
320}
321
322// Assume here that the Option carries an error
323impl<F: ReturnError> HandleError for Option<F>
324{
325  type OkType = ();
326  type ErrorType = F;
327
328  fn handle(self) -> Self::OkType
329  {
330    if let Some(error) = self
331    {
332      error.fatal()
333    }
334  }
335  fn or_warn(self) -> Option<Self::OkType>
336  {
337    self.map(ReturnError::warn)
338  }
339}
340
341impl<S, F: ReturnError> AssessError<S, F> for Result<S, F>
342{
343  fn assess(self, optional: bool) -> Result<Option<S>, F>
344  {
345    match (self)
346    {
347      // We have a value to return here
348      Ok(ok) => Ok(Some(ok)),
349      // Everything is okay but we don't have a value
350      Err(err) if (optional) => { err.warn(); Ok(None) },
351      Err(err) => Err(err)
352    }
353  }
354}

358/// Error type definition in macro to reduce boilerplate
359#[macro_export]
360macro_rules! error_ty
361{
362  {
363    $(#[$attr: meta])*
364    $vis: vis enum Error $(<$($generic: ident $(: $($depend: ty),*)?),*>)?
365    {
366      $(
367        $(#[$v_attr: meta])*
368        $variant: ident $(<$($v_generic: ident $(: $($v_depend: ty),*)?),*>)?
369      ),*
370    }
371  } => {
372    pub use std::result::Result as StdResult;
373
374    /// Result with an error variant that is traced
375    pub type Result<T> = StdResult<T, ErrorTrace>;
376    /// Result with a basic, traceless error
377    pub type BasicResult<T> = StdResult<T, Error>;
378
379    $(#[$attr])*
380    $vis enum Error $(<$($generic $(: $($depend),*)?),*>)?
381    {
382      $(
383        $(#[$v_attr])*
384        $variant $(<$($v_generic $(: $($v_depend),*)?),*>)?
385      ),*
386    }
387
388    // This stores an error (in kind) and an error trace/"subject" (in trace)
389    #[derive(PartialEq, Eq, Clone, Debug)]
390    #[must_use]
391    $vis struct ErrorTrace
392    {
393      pub kind: Error,
394      pub subject: Option<String>,
395      pub trace: String
396    }
397
398    impl<OkType, ErrType: Display> ErrorResult<Error, ErrorTrace, OkType, ErrType> for StdResult<OkType, ErrType>
399    {
400      fn into_trace(self, kind: Error) -> Result<OkType>
401      {
402        self.map_err(|trace| kind.trace(&trace))
403      }
404    }
405
406    impl<ErrType: Display> ErrorResult<Error, ErrorTrace, (), ErrType> for Option<ErrType>
407    {
408      fn into_trace(self, kind: Error) -> Result<()>
409      {
410        if let Some(err) = self.map(|trace| kind.trace(&trace))
411        {
412          Err(err)
413        }
414        else {
415          Ok(())
416        }
417      }
418    }
419
420    impl ErrorResult<Error, ErrorTrace, (), String> for String
421    {
422      fn into_trace(self, kind: Error) -> Result<()>
423      {
424        Err(kind.trace(&self))
425      }
426    }
427
428    impl ErrorResult<Error, ErrorTrace, (), std::io::Error> for std::io::Error
429    {
430      fn into_trace(self, kind: Error) -> Result<()>
431      {
432        Err(kind.trace(&self))
433      }
434    }
435
436    // Strip a traceful error down to a traceless error
437    impl From<ErrorTrace> for Error
438    {
439      fn from(trace: ErrorTrace) -> Self
440      {
441        trace.kind
442      }
443    }
444
445    // ...and vice versa
446    impl From<Error> for ErrorTrace
447    {
448      fn from(kind: Error) -> Self
449      {
450        kind.trace("")
451      }
452    }
453
454    impl Error
455    {
456      pub fn trace(self, trace: impl Display) -> ErrorTrace
457      {
458        ErrorTrace { kind: self, subject: None, trace: trace.to_string() }
459      }
460    }
461
462    impl ErrorTrace
463    {
464      pub fn subject(self, subject: impl Display) -> Self
465      {
466        Self { kind: self.kind, subject: Some(subject.to_string()), trace: self.trace }
467      }
468
469      pub fn option_subject(self, subject: Option<impl Display>) -> Self
470      {
471        Self { kind: self.kind, subject: subject.map(|t| t.to_string()), trace: self.trace }
472      }
473    }
474
475    // If this `Result` is of `Err` variant, add some subject to it
476    impl<OkType> ExtendWithSubject<OkType, ErrorTrace> for Result<OkType>
477    {
478      fn subject(self, subject: impl Display) -> Result<OkType>
479      {
480        match (self)
481        {
482          Ok(ok) => Ok(ok),
483          Err(err) => Err(ErrorTrace { kind: err.kind, subject: Some(subject.to_string()), trace: err.trace })
484        }
485      }
486    }
487  };
488}

491/// Safeguards to throw errors on certain evaluations
492#[macro_export]
493macro_rules! guard
494{
495  ($eval: expr => $error: expr) =>
496  {
497    if ($eval)
498    {
499      return Err($error)
500    }
501  }
502}
