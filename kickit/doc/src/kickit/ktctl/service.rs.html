7/// Frontend handle to init service, sourced from cache & configuration
8/**
9  * # Architecture
10  *
11  * ## Configuration
12  *
13  * Service configurations are stored in `/usr/lib/kickit/service`. However, only some basic information
14  * (like `optional` and `description` fields) are sourced from this file, as important fields (such as
15  * pattern & sandboxing) may be changed after a service has been started, causing an invalid representation
16  * of the service as it is running. Another problem is we want to know fields defined only after execution
17  * (through [`OnceLock`] and such), which cannot be provided by the configuration file.
18  *
19  * ## Cache
20  *
21  * The cache exists to fill in for these 2 aforementioned downfalls of using just the configuration file.
22  * It provides the integral, static fields of `pattern` and sandbox status, as well as the later-assigned
23  * `pid` (for `Standard` services) and `status` (for `RunOnce` services).
24  *
25  * Each service's cache is located in its respective `run` directory, and is globally accessible (as it
26  * does not contain any sensitive information).
27  *
28  * # Limitations
29  *
30  * Absolutely no modifications are perfomed by the frontend. This is by design (see documentation for more
31  * information).
32  */
33#[derive(PartialEq, Eq, Clone, Debug)]
34pub struct ServiceFrontend
35{
36  /// Service name, used to index configuration directory & acquire cache instance
37  pub name: BoxedStr,
38  /// Service execution pattern, sourced from service cache
39  pub pattern: Pattern,
40  /// Sandbox status, sourced from service cache
41  pub sandboxed: bool,
42  // Exit status 
43  status: Option<i32>,
44  pid: u32,
45  description: Option<BoxedStr>,
46  optional: bool
47}

49/// The two basic configuration options from the TOML config file
50#[derive(PartialEq, Eq, Clone, Debug, Default, serde::Deserialize)]
51pub struct ServiceConfig
52{
53  pub description: Option<BoxedStr>,
54  pub optional: Option<bool>
55}
56
57#[derive(PartialEq, Eq, Copy, Clone, Debug)]
58struct CacheInstance
59{
60  pattern: u8,
61  sandboxed: u8,
62  pid: u32,
63  status: Option<i32>
64}
65
66impl CacheInstance
67{
68  // Read 3 seperate bytes from cache, followed by 4 bytes of LE u32 bytes (pid)
69  pub(super) fn new(mut source: impl io::Read) -> Result<Self>
70  {
71    let mut out = [0u8; 7];
72    source.read_exact(&mut out).into_trace(Error::ServiceConfig)?;
73
74    let (bytes, pid_raw, status_switch) = (sized_slice!(out[0..2]), sized_slice!(out[2..6]), out[6] == 1);
75
76    // Check the option switch to see if there is a status available for us
77    let status = status_switch.then_some(
78    {
79      // Exit status of this service in i32 LE bytes
80      let mut status = [0u8; 4];
81      source.read_exact(&mut status).into_trace(Error::ServiceConfig)?;
82      i32::from_le_bytes(status)
83    });
84
85    Ok(Self { pattern: bytes[0], sandboxed: bytes[1], pid: u32::from_le_bytes(pid_raw), status })
86  }
87
88  pub(super) fn export(self) -> Result<(Pattern, bool, u32, Option<i32>)>
89  {
90    // Must be one of the 3 patterns available, see `init::service::Pattern` for their respective bytes
91    let pattern = <Pattern as TryFrom<u8>>::try_from(self.pattern).into_trace(Error::ServiceConfig)?;
92    // This is just a bool so will be 0 for false, 1 for true
93    let sandboxed: bool = self.sandboxed.try_into().into_trace(Error::ServiceConfig)?;
94
95    Ok((pattern, sandboxed, self.pid, self.status ))
96  }
97}

99impl ServiceConfig
100{
101  pub fn try_source(name: &str) -> Result<Self>
102  {
103    let config_toml = std::fs::read_to_string(path!(crate::PREFIX, "service", format!("{name}.toml")))
104                        .into_trace(Error::BadService).subject(name)?;
105
106    toml::from_str(&config_toml).into_trace(Error::ServiceConfig).subject(name)
107  }

110impl ServiceFrontend
111{
112  #[inline]
113  #[must_use]
114  pub fn path(&self, path: &str) -> PathBuf
115  {
116    if (path == "stat")
117    {
118      PathBuf::from(format!("/proc/{}/stat", self.pid))
119    }
120    else {
121      path!("/run/kickit/service", self.name.as_ref(), path)
122    }
123  }
124
125  /**
126    * # Errors
127    *
128    * * Service does not exist for this current init session
129    *
130    * * Failed to open the service's cached configuration file
131    *
132    * * Failed to export cache configuration
133    */
134  pub fn import(name: &str) -> Result<Self>
135  {
136    // Make sure the requested service actually exists for this init session
137    guard!(!path!("/run/kickit/service", name).is_dir() =>
138            Error::BadService.trace("No such file or directory").subject(name));
139
140    // Open the cached config file
141    let cache_file = File::open(path!("/run/kickit/service", name, "config")).into_trace(Error::BadService)
142                        .subject(name)?;
143
144    // Source cache from binary file contents
145    let cache = CacheInstance::new(cache_file).subject(name)?;
146
147    // Convert values from raw byte to their types
148    let (pattern, sandboxed, pid, status) = cache.export().subject(name)?;
149    // Fallback to default if configuration is inaccessible for some reason (no description, not optional)
150    let ServiceConfig { description, optional } = ServiceConfig::try_source(name).unwrap_or_default();
151
152    Ok(Self { name: name.into(), pattern, sandboxed, status, pid, description, optional: optional == Some(true) })
153  }
154
155  /// Print this service's information in a tree-style formatted output
156  /**
157    * # Output
158    *
159    * ```text
160    * +- standard_service_example
161    *  ├─ Status:       up
162    *  ├─ PID:          6767
163    *  ├─ Optional:     yes
164    *  └─ Description:  An example standard service
165    *
166    * ```
167    *
168    * # Errors
169    *
170    * * Failed to get the proccess's stat path
171    *
172    * * Failed to get the service's PID
173    *
174    * # Panics
175    *
176    * * `Service::path` failed to get the stat file (should never happen)
177    */
178  pub fn pretty_print(mut self) -> Result<()>
179  {
180    // The "root" of the tree, just this service's name/identifier
181    println!("+- {}", Colourize::bold(&self.name));
182
183    // Descriptions are optional, but most services should ideally provide these
184    if let Some(desc) = self.description.take() { println!("  ├─ Description:  {desc}"); }
185    // Optional services are uncommon, so a `optional: no` field isn't included here
186    if (self.optional) { println!(" ├─ Optional:     yes"); }
187
188    let status = if (self.pattern == Standard)
189    {
190      if (self.sandboxed) { println!(" ├─ Sandboxed:    {}", "yes".colour(Colour::Green)); }
191
192      println!(" ├─ PID:          {}", self.pid);
193
194      // Read the process's stat file from `/proc/<PID>/stat`, and then get the 3rd field, which is its running state
195      let stat = fs::read_to_string(self.path("stat")).map(|c| c.split_whitespace().nth(2).map_to_owned());
196
197      match (pat_layer_tree!(stat => Ok => Some).map_as_str())
198      {
199        // These are all acceptable process statuses (S = sleeping, I = idle, D = disk sleep, R = running)
200        Some("S" | "I" | "D" | "R") => Colourize::colour("up", Colour::Green),
201        // Z = zombie (stopped running) and X = killed (by another process)
202        Some("Z" | "X") => Colourize::colour("down", Colour::Red),
203        // Usually after a process exits on its own accord, its proc directory is deleted
204        _ => Colourize::colour("dead", Colour::Red)
205      }
206    }
207    else {
208      match (self.status)
209      {
210        // Exited on OK status (code 0)
211        None | Some(0) => Colourize::colour("finished", Colour::Green),
212        // Process failed on error code (0 <)
213        Some(err_code) => format!("failed (code {err_code})").colour(Colour::Red)
214      }
215    };
216    println!(" └─ Status:       {status}");
217
218    // Print newline seperator for next service
219    println!();
220
221    Ok(())
222  }

224  /**
225    * # Errors
226    *
227    * * Service's log file is not accessible (this should never be the case)
228    *
229    * * Encountered log entry with an invalid timestamp (should be in Unix millis)
230    *
231    * * Timestamp parsing failure in [`from_timestamp_millis`]
232    *
233    * * `stdout` console is not writable (TTY error, not kickit related)
234    *
235    * [`from_timestamp_millis`]: chrono::DateTime::from_timestamp_millis
236    */
237  pub fn logs(&self, ugly: bool, ignore_init: bool) -> Result<()>
238  {
239    use std::{io::{BufReader, Write, stdout, ErrorKind}, process::exit};
240    use crate::{DumpVec, DequeData, DelimReader, init::service::logger::LogEntry};
241    // Wrapper decoder over a readable type
242    use ruzstd::decoding::StreamingDecoder;
243    use time_format::format_common_utc;
244
245    // Open the zstd compressed log file
246    let file = File::open(self.path("log")).into_trace(Error::BadService).subject(&self.name)?;
247    // The decoder implements Read to idiomatically decompress
248    let decoder = StreamingDecoder::new(file).into_trace(Error::LogAccessFail).subject(&self.name)?;
249    // Use `BufReader` for the handy `read_until` method
250    let log = BufReader::new(decoder);
251
252    // Obtain a direct writable instance to the `stdout` for SIGPIPE handling (more info below)
253    let mut out = stdout().lock();
254
255    /*
256     * We don't uses the `lines()` method here because that creates a String but our entries contain binary data
257     * so we need a vector of bytes instead
258     */
259    for mut entry in (DelimReader::new(log, b'\n').map(<DequeData as From<Data>>::from))
260    {
261      // First 4 bytes are i32 seconds timestamp (time elapsed since Unix epoch) in little-endian order
262      let timestamp_bytes: [u8; 4] = entry.dump_front();
263      // Represent the timestamp as was supposed to be, a real integer
264      let timestamp: i64 = i32::from_le_bytes(timestamp_bytes).into();
265
266      // Init entries are marked by an extra byte after timestamp, before UTF-8 string contents
267      let from_init = entry[0] == LogEntry::INIT_ENTRY;
268
269      if (from_init)
270      {
271        continueif! (ignore_init);
272        // Remove this byte now that we have parsed it
273        entry.pop_front();
274      }
275
276      // Included after timestamp in bold to identify an init entry
277      let marker = from_init.then_some(tern! { ugly => "(kickit) ", _ => "\x1b[1m(kickit)\x1b[0m " })
278                      .unwrap_or_default();
279
280      // `entry` isn't used after this, so this resets it for the next iteration
281      let contents = String::from_utf8(std::mem::take(&mut entry).into()).into_trace(Error::Format)?;
282
283      let line = tern!
284      {
285        // No colour formatting for these entries
286        ugly => format!("[{timestamp}] {marker}{contents}\n"),
287        _ => {
288          let time_fmt = format_common_utc(timestamp, time_format::DateFormat::Custom("%d %b %Y, %H:%M:%S"))
289                          .into_trace(Error::Time).subject(&self.name)?;
290
291          format!("[{time_fmt}] {marker}{contents}\n")
292          
293        }
294      };
295
296      /*
297       * When we output to stdout and another program is piped (e.g. grep), the program cuts off the pipe
298       * after it has consumed everything it needs so we end up with SIGPIPE. We don't error here if found,
299       * since this is expected & not an actual error, though is treated as one by `println!` so we have to
300       * do a janky workaround for compatibility
301       *
302       * (note): This may change in the future though, see <https://github.com/rust-lang/rust/issues/62569>
303       */
304      match (out.write_all(line.as_bytes()))
305      {
306        // Exit normally without throwing an error on SIGPIPE
307        Err(sig_pipe) if (sig_pipe.kind() == ErrorKind::BrokenPipe) => exit(0),
308        // OK result or an actual error
309        t => t
310      }
311        .into_trace(Error::Format)?;
312    }
313    Ok(())
314  }
315}

317/// List all services with [`ServiceFrontend::print`], or only the ones in `targets`, if provided
318/**
319  * # Errors
320  *
321  * * Failed to open the service from its name (see `Service::From::<&str>::from`)
322  *
323  * * Failed to print the service's details (pid & status)
324  */
325pub fn serviceList(targets: Option<&[String]>) -> Result<()>
326{
327  for service_res in (fs::read_dir(PathBuf::from("/run/kickit/service")).into_trace(Error::RunFsParseFail)?)
328  {
329    let service = service_res.into_trace(Error::BadService)?.file_name();
330
331    // Import each entry as a service from its OsString file name
332    if let Some(up_service) = ServiceFrontend::import(service.display().to_string().as_str()).or_warn()
333    {
334      let name = &*up_service.name;
335
336      // Only print if targets provided by user allow
337      if (targets.is_none_or(|targets| targets.contains(&name.into())))
338      {
339        up_service.pretty_print()?;
340      }
341    }
342  }
343
344  Ok(())
