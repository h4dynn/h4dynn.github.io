14// Interpret given response from the socket as Result enum
15trait SocketResult<OkType>
16{
17  fn result(&self) -> Result<StdResult<OkType, socket::Error>>;
18}
19
20pub trait Request: Socket + Sized + Send + Sync
21{
22  // Expect the socket to give us a response
23  fn request(self, input: u8) -> impl Future<Output = Result<StdResult<Data, socket::Error>>> + Send
24  {
25    use tokio::{net::UnixStream, io::AsyncReadExt};
26
27    async move {
28      // Determine where the socket is that we want to interact with
29      let path = self.path();
30
31      // We don't know how long the reply is going to be, so a Vec<u8> works best
32      let mut out = Data::new();
33      // Open a new connection to the socket, may fail if there is an existing connection
34      let mut io = UnixStream::connect(&path).await.into_trace(Error::SocketAccessFail)
35                                                      .subject(path.display())?;
36
37      // Wait until the stream is ready to accept being written to
38      io.writable().await.into_trace(Error::SocketAccessFail)?;
39      io.try_write(&[input]).into_trace(Error::AccessRunFsFail)?;
40
41      // ..and then wait for the stream to have a reply available
42      io.readable().await.into_trace(Error::SocketAccessFail)?;
43      io.read_to_end(&mut out).await.into_trace(Error::SocketAccessFail).subject("io.Core")?;
44
45      out.result()
46    }
47  }
48}
49
50impl<S: AsRef<[u8]>> SocketResult<Vec<u8>> for S
51{
52  fn result(&self) -> Result<StdResult<Vec<u8>, socket::Error>>
53  {
54    let inner = self.as_ref();
55
56    if (inner.len() < 2)
57    {
58      return Err(Error::SocketResponse.trace(format!("Expected at least 2 bytes, got {}", inner.len())))
59    }
60
61    match (inner[0])
62    {
63      socket::Error::OK => Ok(Ok(inner[1..].to_vec())),
64      // Match the returned byte to its corresponding error variant
65      socket::Error::ERR => Ok(Err(socket::Error::errorize(inner[1]).into_trace(Error::SocketResponse)?)),
66      // This shouldn't ever happen unless there is a bug in the init itself
67      bad => Err(Error::SocketResponse.trace(format!("Expected ok/err, got {bad}")))
68    }
69  }
70}
