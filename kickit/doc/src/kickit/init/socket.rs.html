6/**
7  * The [`Open::open_sock`] method is a function with a blanket which keeps the implementation universal
8  * between all sockets and stops custom implementations
9  */
10pub trait Open: Socket + Sync + 'static
11{
12  /// Initializes the socket in its provided path, creating a listener for input/output
13  fn open_sock(&'static self) -> impl Future<Output = Result<()>> + Send
14  {
15    async move
16    {
17      use std::{os::unix::fs::PermissionsExt, fs::{Permissions, set_permissions}};
18      use tokio::{task, net::UnixSocket};
19
20      let path = self.path();
21      let permissions = Permissions::from_mode(if Self::PRIVATE { 0o600 } else { 0o666 });
22
23      let sock = UnixSocket::new_stream().into_trace(Error::SocketStartup)?;
24      // Bind (start) our socket here
25      sock.bind(&path).into_trace(Error::RunFsFail).subject(self.path().display())?;
26
27      /*
28       * thanks <https://users.rust-lang.org/t/how-to-manage-permissions-of-a-unixlistener/31039/8>
29       * for having an answer for this it really hurt my head
30       */
31      set_permissions(path, permissions).into_trace(Error::ServiceAccess).subject(self.path().display())?;
32
33      // Set max listeners at a time (backlog)
34      let listener = sock.listen(Self::MAX_LISTENERS).into_trace(Error::SocketIo)?;
35
36      // Loop forever, handling each incoming connection
37      loop {
38        // Start accepting incoming connections
39        let (mut socket, _) = listener.accept().await
40                                .map_err(|_| Error::SocketStartup.trace("failed to start listener"))?;
41
42        task::spawn(async move { self.handler(&mut socket).await; });
43      }
44    }
45  }
46}

48/// Relay to the receiving peer of this socket a `Result` in byte form (see [`crate::socket::Error`])
49#[macro_export(local_inner_macros)]
50macro_rules! socket_relay
51{
52  // Provide the OK byte as well as our bytes
53  ($stream: expr, Ok($ok: expr)) =>
54  {{
55    socket_relay!(@__Write $stream, &[socket::Error::OK as u8]);
56    socket_relay!(@__Write $stream, $ok);
57  }};
58  /*
59   * Provide a failure byte to the socket peer as a signal that something went wrong, shutdown the
60   * connection & then return, see `socket::Error`for all possible errors
61   */
62  ($stream: expr, Err($error: path)) =>
63  {{
64    // Write our "error byte" to signal to peer an error has occurred
65    $stream.try_write(&[socket::Error::ERR, $error as u8]).into_trace(Error::SocketIoWrite).or_warn();
66    // Exit our function- do nothing more here
67    return
68  }};
69  (@__Write $stream: expr, $bytes: expr) =>
70  {
71    if let Err(err) = $stream.try_write($bytes).into_trace(Error::SocketIoWrite)
72    {
73      // TO-DO: Generally for debugging purposes only, so may fill up init logs if we're not careful
74      err.warn();
75
76      socket_relay!($stream, Err(socket::Error::IoWrite))
77    }
78  };
79}

82#[macro_export]
83macro_rules! socket_guard
84{
85  ($stream: expr, $eval: expr => $error: path) =>
86  {
87    if ($eval)
88    {
89      $crate::socket_relay!($stream, Err($error));

95#[macro_export]
96macro_rules! socket_receive
97{
98  ($stream: expr, $buf_size: tt) =>
99  {{
100    let mut __buf__ = [0u8; $buf_size];
101
102    $crate::socket_guard!($stream, $stream.try_read(&mut __buf__).is_err() => socket::Error::IoRead);
103
104    __buf__
105  }};
106}

108/// Make sure init stream ([`UnixStream`]) is ready to be read/written to
109#[macro_export]
110macro_rules! socket_capability_check
111{
112  ($Self: ident, $stream: expr => Listen) =>
113  {
114    $crate::socket_guard!($stream, $stream.readable().await.is_err() => socket::Error::NotReadReady)
115  };
116  ($Self: ident, $stream: expr => Relay) =>
117  {
118    if ($stream.writable().await.is_err())
119    {
120      // No error is relayed back to the peer, as we cannot write back
121      Error::SocketIoWrite.trace("peer is refusing writes").subject($Self::NAME).warn();
122      return
123    }
124  };
125  ($Self: ident, $stream: expr => Listen + Relay) =>
126  {
127    $crate::socket_capability_check!($Self, $stream => Listen);
128    $crate::socket_capability_check!($Self, $stream => Relay);
129  };
130}
131pub use crate::socket_capability_check as capability_check;

133impl Socket for Core
134{
135  const NAME: &str = "Core";
136  // All users should be able to access this socket as it reports no private data
137  const PRIVATE: bool = false;
138
139  async fn handler(&self, stream: &mut UnixStream)
140  {
141    use crate::{state::state, init::target::TARGET};
142    use std::process::id as pid;
143
144    capability_check!(Self, stream => Listen + Relay);
145
146    match (socket_receive!(stream, 1))
147    {
148      [Self::STATE] => relay!(stream, Ok(&[state!() as u8])),
149      // Add a newline byte, this is our EOL
150      [Self::VERSION] => relay!(stream, Ok(&[env!("CARGO_PKG_VERSION").as_bytes(), b"\n"].concat())),
151      [Self::TARGET] =>
152      {
153        if let Some(target) = TARGET.get()
154        {
155          relay!(stream, Ok(&[target.name.as_bytes(), b"\n"].concat()));
156        }
157        else {
158          // Error if target name cannot be reached for whatever reason
159          relay!(stream, Err(socket::Error::Internal));
160        }
161      },
162      [Self::PID] => relay!(stream, Ok(&pid().to_le_bytes())),
163      // Safely ignore newlines
164      [b'\n'] => (),
165      // Send an error for unknown bytes
166      _ => relay!(stream, Err(socket::Error::BadInput))
167    }
168  }
169}

171impl Socket for Log
172{
173  const NAME: &str = "Log";
174
175  async fn handler(&self, stream: &mut UnixStream)
176  {
177    use crate::init::console::SUPER_LOG;
178
179    capability_check!(Self, stream => Listen + Relay);
180
181    if socket_receive!(stream, 1) == [Self::MASTER] && let Ok(log) = SUPER_LOG.lock()
182    {
183      // The log is a vector of strings, so we seperate each member by a newline
184      relay!(stream, Ok(log.join("\n").as_bytes()));
185    }
186    else {
187      // Cannot acquire a lock on the `SUPER_LOG`. Generally, this should never occur.
188      relay!(stream, Err(socket::Error::Internal));
189    }
190  }
191}
