14/// Power signal recipient & handler socket
15/**
16  * # Function
17  *
18  * Receives poweroff/restart signals (which may also be forceful). After receiving a signal, it
19  * begins the poweroff routine (see [`poweroff`] function).
20  *
21  * This socket is private so that these signals can only be sent by the root user.
22  *
23  * # Thread
24  *
25  * Like every other socket, its handler is ran on the init's thread. However, differently to
26  * the other sockets its job is specific to the init's execution itself.
27  */
28#[derive(PartialEq, Eq, Copy, Clone, Default, Debug)]
29pub struct PowerInt;
30
31/// Poweroff the system gracefully, performing required operations in a routine order
32/**
33  * # Routine
34  *
35  * The tasks performed by this method, to ensure graceful shutdown, in order:
36  *
37  * 1. The `OnceLock` [`POWER_OFF_READY`] is initialized to `true`. This value is checked by
38  *    the service supervisor after a service is killed/downed. Setting it tells the
39  *    supervisor that the service has been downed as part of the poweroff process, and as
40  *    such to not throw an error,
41  *
42  * 2. Each service in the global `OnceLock` [`STANDARD_SERVICES`] is [`down`]ed,
43  *
44  * 3. Each mount entry in `/proc/mounts` is unmounted (except for critical system pseudo
45  *    filesystems, like proc, dev & sys),
46  *
47  * 4. The Linux system call `reboot` is called, with the relevant mode, and the kernel
48  *    handles the rest of the poweroff process.
49  *
50  * ## Non-init process
51  *
52  * When `kickit` is not ran as the init process, the behavior of this method is very
53  * different:
54  *
55  * * First off, only mount entries from the `/run/kickit` directory are unmounted. This is
56  *   because the other mount entries (most likely sourced from `/etc/fstab`) are handled by
57  *   the actual init system, so we don't want to touch them.
58  *
59  * * Secondly, the `/run/kickit` directory is deleted. This is not relevant when powering-off
60  *   as the init, since the `run` temporary filesystem will be unmounted anyways.
61  *
62  * * And finally, the `kickit` process is exited (no poweroff; again, this is the actual
63  *   init's job & not ours).
64  *
65  * # Circumstances
66  *
67  * This function should only ever be 
68  *
69  * # Errors
70  *
71  * * Failed to get the `init::SERVICE_WATCHERS` oncelock since it isn't set
72  *
73  * * Failed to read from /run/kickit/service
74  *
75  * * Failed to parse the PID file as it isn't numeric
76  *
77  * * Failed to kill a service
78  *
79  * * Failed to reboot with `nix::sys::reboot::reboot`
80  *
81  * [`STANDARD_SERVICES`]: crate::init::service::STANDARD_SERVICES
82  * [`down`]: crate::init::service::Service::down
83  */
84#[cold]
85pub fn poweroff(restart: bool /* this is referred to as "autoboot" by the syscall */) -> Result<SystemPowerOff>
86{
87  use crate::{init::{init_guard, mount::{mounttab, unmount_flags}, service::STANDARD_SERVICES}};
88  use std::{fs, path::PathBuf, process};
89
90  // Match a path-referable type to any one of the provided right-hand side paths
91  macro_rules! any_path
92  {
93    ($lhs: expr, $($rhs: literal)|*) =>
94    {
95      $($lhs.as_path() == <std::path::Path>::new($rhs))||*
96    };
97  }
98
99  // This makes sure that when we kill the services, the service manager doesn't throw an error
100  oncelock! { POWER_OFF_READY = true }?;
101
102  // If an error has occurred during service startup, we need to still be able to accept poweroff calls without error
103  if let Some(services) = STANDARD_SERVICES.get()
104  {
105    for service in services
106    {
107      log(Status(format!("Killing service: {}", service.cfg.name)));
108      debug_cc!(&service);
109
110      // This is the only instance where the down method should ever be called
111      service.down()?;
112    }
113  }
114  else {
115    log(Warning("Standard services have not been initialized; skipping downing services"));
116  }
117
118  for entry in (mounttab()?)
119  {
120    /*
121     * Do not touch any other mountpoints outside of kickit's scope if we are not the init process.
122     * We still loop through each entry regardless, since service sandboxing will create bind mounts
123     * and system VFS mounts, which need to be removed before the directory can be deleted.
124     */
125    continueif! (init_guard() && !entry.dest().starts_with("/run/kickit/"));
126
127    // These are special system filesystems, unmounting them would be bad
128    if (!any_path!(entry.dest(), "/dev" | "/dev/pts" | "/proc" | "/sys" | "/sys/fs/cgroup" | "/run"))
129    {
130      log(Status(format!("Unmounting filesystem: {}", entry.dest().display())));
131
132      // Attempt regular unmount (without any flags) first before lazy unmount
133      if (entry.unmount(None).or_warn().is_none())
134      {
135        log(Warning("Failed to unmount filesystem, it will be lazily unmounted"));
136
137        // Last resort- obviously not ideal but we still need to unmount
138        entry.unmount(Some(unmount_flags![Lazy])).or_warn();
139      }
140    }
141  }
142
143  /*
144   * Perform a seperate procedure for when kickit is ran as a regular process & not the init,
145   * we don't actually poweroff the system in this case, just kill the init
146   */
147  if (init_guard())
148  {
149    log(Status("Stopping kickit now"));
150    // This is useful for when debugging & running kickit multiple times after new builds
151    fs::remove_dir_all(PathBuf::from("/run/kickit")).into_trace(Error::Shutdown)?;
152    // ...and we're done! ^_^
153    process::exit(0);
154  }
155
156  // We have completed the routine so we can call the simplistic syscall
157  unsafe { force_poweroff(restart) }
158}

160/// Skip normal poweroff routine, and shutdown immediately (**not recommended!**)
161/**
162  * # Errors
163  *
164  * This function is a frontend to [`nix::sys::reboot::reboot`]. Refer to its error documentation.
165  *
166  * # Safety
167  *
168  * This function is marked `unsafe` due to its lack of a safe poweroff routine, skipping straight
169  * to the basic Linux `reboot` syscall.
170  *
171  * Resulting potential issues include:
172  *
173  * ## Data loss
174  *
175  * In the regular poweroff routine, each filesystem is unmounted, allowing them to sync their data
176  * properly & in a safe manner. As this is skipped, they have no opportunity to do so, leading to
177  * potential data loss.
178  *
179  * This risk is mitigated by [journaling], which is performed by the vast majority of modern
180  * filesystems[^1]. Journaling filesystems keep track of uncommitted modifications, before they
181  * complete, reducing the risk of corruption after being brought back online.
182  *
183  * However, after being brought back online, if there was any data being actively written to the
184  * disk, it may become lost or corrupted.
185  *
186  * **TL;DR:** Journaling will save your filesystem *majority* of the time, but don't rely on it
187  * exclusively, and ideally have a backup
188  *
189  * [^1]: incomplete list of common filesystems with journaling feature: ext3, ext4, f2fs, xfs,
190  * btrfs, bcachefs, zfs, hfsplus, ntfs.
191  *
192  * ## 
193  *
194  * [journaling]: <https://en.wikipedia.org/wiki/Journaling_file_system>
195  */
196#[cold]
197pub unsafe fn force_poweroff(restart: bool) -> Result<SystemPowerOff>
198{
199  let mode = if (restart)
200  {
201    RebootMode::RB_AUTOBOOT
202  }
203  else {
204    RebootMode::RB_POWER_OFF
205  };
206
207  // You can see from how this function is literally just a single call, why this really isn't recommended
208  nix::sys::reboot::reboot(mode).into_trace(Error::PowerCritical)
209}

211impl PowerInt
212{
213  /// Standard shutdown request
214  pub const SHUTDOWN: u8 = 0xe0;
215  /// Standard reboot request
216  pub const REBOOT: u8 = 0xd7;
217  /// Bypass checks & graceful processes, skipping straight to shutdown (not recommended)
218  pub const FORCE_SHUTDOWN: u8 = 0xfd;
219  /// Bypass checks & graceful processes, skipping straight to reboot (not recommended)
220  pub const FORCE_REBOOT: u8 = 0xe3;
221}
222
223impl Socket for PowerInt
224{
225  const NAME: &str = "PowerInt";
226
227  async fn handler(&self, stream: &mut UnixStream)
228  {
229    use crate::{socket, init::{init_guard, socket::{relay, capability_check}, power::{poweroff, force_poweroff}}};
230
231    // Unique to this socket, we don't send any replies back, we only listen
232    capability_check!(Self, stream => Listen);
233
234    let mut input = [0u8];
235
236    if (stream.try_read(&mut input).is_err())
237    {
238      relay!(stream, Err(socket::Error::IoRead));
239    }
240    debug_cc!(&input[0]);
241
242    match (input[0])
243    {
244      Self::SHUTDOWN => poweroff(false).or_warn(),
245      Self::REBOOT => poweroff(true).or_warn(),
246      Self::FORCE_SHUTDOWN | Self::FORCE_REBOOT if init_guard() =>
247      {
248        Error::Shutdown.trace("force shutdown expects kickit to be the init process (1)").warn();
249        relay!(stream, Err(socket::Error::Unsupported));
250      },
251      Self::FORCE_SHUTDOWN => unsafe { force_poweroff(false) }.or_warn(),
252      Self::FORCE_REBOOT => unsafe { force_poweroff(true) }.or_warn(),
253      // Write error byte to socket- unexpected input
254      _ => relay!(stream, Err(socket::Error::BadInput))
255    };
256  }
