1/*!
2 * # Service lifetime control
3 *
4 * This module includes implementations for supervising standard services. By default, the [Root]
5 * lifetime is used. For user services, a seperate [User] lifetime is used
6 */
7
8use std::future;
9use crate::{AsyncBlock, init::{console::prelude::*, service::{GenericService, Supervisor}, power::POWER_OFF_READY}};
10
11/// Enforcement rules for service process supervision
12pub trait ServiceLifetime
13{
14  /**
15    * Supervise a daemon-like service ([Standard]) to make sure it never dies
16    *
17    * [Standard]: super::Pattern::Standard
18    */
19  fn supervise(supervisor: &mut Supervisor) -> impl Future<Output = Result<AsyncBlock>> + Send;
20}
21
22/**
23  * # Root service lifetime
24  *
25  * Used for services executed at the init-level, as the root user. Standard services are supervised
26  * to forbid sudden exits. On exit, an error will be thrown (except for power-off).
27  *
28  * ## Design
29  *
30  * ### Example standard service (supervisor task)
31  *
32  * Simplified code:
33  *
34  * ```rust
35  * // Block until the process returns
36  * supervisor.process.wait().await;
37  *
38  * // On power-off, the process is expected to return, so prevent an error being thrown
39  * if (!POWER_OFF_READY)
40  * {
41  *   return Err(Error::ServiceDown.trace("service died during supervision").subject(supervisor.name))
42  * }
43  * ```
44  */
45#[derive(PartialEq, Eq, Copy, Clone, Debug, Default, derive_more::Display)]
46pub struct Root;
47
48/**
49  * # User service lifetime
50  *
51  * As these services are ran under the user's profile, not root, we can be alot more leanient-
52  * and allow these services to be killed at anytime by the user.
53  */
54#[derive(PartialEq, Eq, Copy, Clone, Debug, Default, derive_more::Display)]
55pub struct User;
56
57impl ServiceLifetime for Root
58{
59  /**
60    * # Errors
61    *
62    * * Service process exited on a code (even if the code is 0)
63    *
64    * * Failed to wait for the process
65    */
66  fn supervise(supervisor: &mut Supervisor) -> impl Future<Output = Result<AsyncBlock>>
67  {
68    // This method will cause this function to block until process death
69    let res = supervisor.process.wait();
70
71    if (POWER_OFF_READY.get() != Some(&true))
72    {
73      return future::ready(Err(match (res.map(|status| status.code()))
74      {
75        Ok(Some(c)) => Error::ServiceDied.trace(format!("exited on code {c}")),
76        Ok(None) => Error::ServiceDied.trace("no code is available"),
77        // Extremely unlikely to happen but we still account for it just in case
78        Err(error) => Error::Unknown.trace(format!("service process supervision error: {error}"))
79      }
80        .subject(&supervisor.name)))
81    }
82
83    loop { std::thread::park() }
84  }
85}
