2 * Provide service configuration & fields in a globally accessible cache file

5use std::{sync::OnceLock, process::ExitStatus};
6use super::{Service, ServiceBody, Pattern};
7
8pub struct Cache
9{
10  pattern: Pattern,
11  sandboxed: bool,
12  code: Option<i32>,
13  pid: Option<u32>
14}
15
16impl Cache
17{
18  pub const fn new(pattern: Pattern, sandboxed: bool, code: Option<i32>, pid: Option<u32>) -> Self
19  {
20    Self { pattern, sandboxed, code, pid }
21  }
22
23  /// Construct portable binary form of the cache instance
24  pub fn as_data(&self) -> [u8; 11]
25  {
26    let mut dat = [0u8; 11];
27
28    (dat[0], dat[1]) = (self.pattern as u8, self.sandboxed.into());
29    // Process ID (pid) is 4 bytes long as a u32
30    index_mut_slice!{ dat[2..6] = self.pid.unwrap_or_default().to_le_bytes(); }
31
32    if let Some(int_state) = self.code
33    {
34      // 0x01 indicates we have a status available - 0x00 indicates none available
35      dat[6] = 0x01;
36      // Exit code is 4 bytes long as an i32
37      index_mut_slice!{ dat[7..11] = &int_state.to_le_bytes(); }
38    }
39    else {
40      // Pad the remainder with null bytes for size consistency when parsing
41      dat[7..11].fill_with(|| 0);
42    }
43    dat
44  }
45}
46
47impl Service
48{
49  /// Construct the cache instance by copying service fields
50  pub(crate) fn cache(&self) -> Cache
51  {
52    use ServiceBody::{RunOnce, Standard};
53
54    Cache::new(self.cfg.pattern, self.cfg.sandbox.is_some(),
55                  // RunOnce services provide an exit status code
56                  pat_layer_tree!(&self.body => RunOnce { code } => |Some| OnceLock::get => |Some| ExitStatus::code),
57                  // Standard services provide a process ID of the active child process
58                  pat_layer_tree!(&self.body => Standard { pid } => |Some| OnceLock::get).copied())
59  }
60}
