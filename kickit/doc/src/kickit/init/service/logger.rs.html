1//! Standardized service logging, with asynchronous piping or basic relaying
2
3use std::{io, fs, sync::Arc};
4use crate::{BoxedStr, AnyStr, init::console::prelude::*};
5
6#[derive(Debug)]
7pub struct Logger
8{
9  pub name: Arc<BoxedStr>,
10  shout: bool,
11  // The current line count
12  line: usize,
13  /// Matching log file
14  pub(super) file: fs::File,
15  /// The read pipe from the service process, set when `up` is ran
16  pub(super) reader: Option<io::PipeReader>
17}
18
19/// Log entry to append to service logfile  with an identified source
20/**
21  * # Binary
22  *
23  * Init entries are additionally appended with a singular [`INIT_ENTRY`] byte to symbolize their origin.
24  * The reader of the log file simply checks if the next byte past the timestamp (11 bytes) is the exact
25  * match of the init entry marker. Padding with null bytes is additionally unnecessary, as the data
26  * after the header's end is zstd-compressed and so will have fixed identifiable magic
27  *
28  *
29  * [`INIT_ENTRY`]: #const.INIT_ENTRY
30  */
31#[derive(PartialEq, Eq, Clone, Debug)]
32pub enum LogEntry<'entry>
33{
34  /// Regular entry from service's stderr & stdout [`std::io::PipeReader`]
35  Service(&'entry str),
36  /// Special entry from init
37  Init(&'entry str)
38}
39
40impl Logger
41{
42  /// Empty zstd file, created from /dev/null (`$ zstd -1c < /dev/null | xxd`)
43  pub const NULL_ZSTD: [u8; 13] = [0x28, 0xb5, 0x2f, 0xfd, 0x24, 0, 0x01, 0, 0, 0x99, 0xe9, 0xd8, 0x51];
44
45  #[must_use]
46  pub const fn new(name: Arc<BoxedStr>, shout: bool, file: fs::File) -> Self
47  {
48    Self { name, shout, line: 0, file, reader: None }
49  }
50
51  /// Initialize a new logger instance for a service
52  /**
53    * # Errors
54    *
55    * This methods expects that the log file has not yet been created, and that the service directory
56    * is accessible & writable
57    */
58  pub fn create(name: Arc<BoxedStr>, shout: bool /* append entry to init master log */) -> Result<Self>
59  {
60    use std::{fs::{File, Permissions}, io::Write, os::unix::fs::PermissionsExt};
61
62    // Create the empty log file
63    let mut log_file = File::options().read(true).write(true).create_new(true)
64                        .open(path!("/run/kickit/service", &**name, "log"))
65                        .into_trace(Error::ServiceAccess).subject(&name)?;
66
67    // Make log file inaccessible to anybody but the root user
68    log_file.set_permissions(Permissions::from_mode(0o600)).into_trace(Error::ServiceAccess)?;
69
70    /*
71     * The log file is going to be decompressed later on to view its contents, the decompressor will fail
72     * if the file is empty so this provides some valid but empty zstd data
73     */
74    log_file.write_all(&Self::NULL_ZSTD).into_trace(Error::ServiceAccess)?;
75
76    Ok(Self::new(name, shout, log_file))
77  }
78
79  /// Watch the reader for the process for updates & then send them to the service's log
80  /**
81    * # Implementation
82    *
83    * This function runs in the following order:
84    *
85    * 1. The `reader` instance ([`PipeReader`]) is taken. This is the stream responsible for piping
86    *    `stderr` and `stdout` output into a singular read-only stream,
87    *
88    * 2. Block until a new entry is received from the reader instance (see [`read_until`]),
89    *
90    * 3. The new entry is appended to the service's log file in its binary format (see [`log`]), with
91    *    current timestamp (Unix epoch, resolved with `chrono` crate) and compressed with ZSTD.
92    *
93    * # Errors
94    *
95    * This method calls for 2 conditions to be met:
96    *
97    * 1. This logger is asynchronous (for standard services, differentiating factor is the reader),
98    *
99    * 2. The service is up & running (handled by supervisor, required for the reader endpoint).
100    *
101    *
102    * [`PipeReader`]: std::io::PipeReader
103    * [`read_until`]: std::io::BufRead::read_until
104    * [`log`]: #method.log
105    */
106  pub fn watch(&mut self) -> Result<()>
107  {
108    use std::{io::{BufRead, BufReader}, mem::take};
109
110    // Standard pipe reader for relaying log output
111    let log_ro = self.reader.take().ok_or(Error::Unknown.trace("logger missing").subject(&*self.name))?;
112    // Using a `BufReader` here is more efficient, as we read line-by-line (with `read_until`)
113    let mut log_buf_ro = BufReader::new(log_ro);
114
115    // Buffer of all the (hopefully) UTF-8 bytes this service will provide
116    let mut received = Vec::new();
117
118    loop {
119      // Read until newlines, capturing contents line-by-line
120      log_buf_ro.read_until(b'\n', &mut received).into_trace(Error::ServiceLogContent)?;
121      // Convert from raw UTF-8 bytes to a string
122      let relay = String::from_utf8(take(&mut received)).into_trace(Error::ServiceLogContent)?;
123
124      for line in (relay.lines())
125      {
126        // And finally add the logs from the service!
127        self.log(&LogEntry::Service(line))?;
128      }
129    }
130  }
131
132  /// Append a new line to the service's log
133  /**
134    * # Errors
135    *
136    * * Couldn't open the service's logfile
137    *
138    * * Couldn't compress or decompress the service's logfile
139    *
140    * * Couldn't open a lock on the file (or unlock)
141    *
142    * * Couldn't write to the logfile
143    *
144    * # Panics
145    *
146    * * More than one line was provided to the function
147    */
148  // TO-DO: current method is to decompress & then recompress, which is kinda inefficient
149  pub fn log(&mut self, entry: &LogEntry<'_>) -> Result<()>
150  {
151    use std::io::{Read, Seek, Cursor};
152    use crate::{state::state, init::console::log};
153    use ruzstd::{decoding::StreamingDecoder, encoding::{compress, CompressionLevel}};
154
155    let (new, from_init) = (entry.value(), matches!(entry, LogEntry::Init(_)));
156
157    // Don't send empty lines if they are found for whatever reason
158    if (new.as_str_ref().is_empty())
159    {
160      return Ok(())
161    }
162
163    /*
164     * We cannot have more than 1 line of content, this function is designed to take a-line-at-a-time
165     * (use `for line in input.split('\n') { ... }`). This is a panic & not an error because it is solely
166     * down to how the program is coded, not user error
167     */
168    assert!(new.as_str_ref().split('\n').count() <= 1, "log() was given more than 1 line of input!");
169
170    /*
171     * Because this function will be called multiple times, changing the seek on the log file, we need to
172     * rewind to the beginning of the file
173     */
174    self.file.rewind().into_trace(Error::ServiceLogContent)?;
175
176    // Get timestamp (# of seconds elapsed since Unix epoch) and size down to i32 (doesn't matter until 2038)
177    #[allow(clippy::cast_possible_truncation)]
178    let time = time_format::now().into_trace(Error::Time)? as i32;
179
180    // The decoder wraps the file's contents & decompresses on read (lazy type)
181    let mut decoder = StreamingDecoder::new(&self.file).into_trace(Error::ServiceLogCompress)?;
182    // Log entry contents to be written to file
183    let mut out = Vec::new();
184
185    // Decompress previous log contents into output vector
186    decoder.read_to_end(&mut out).into_trace(Error::ServiceLogCompress)?;
187
188    // Send timestamp to log in little-endian bytes (i32 is 4 bytes long)
189    out.extend_from_slice(&time.to_le_bytes());
190
191    // This byte is how ktctl can differentiate messages from init & service
192    if (from_init)
193    {
194      out.push(LogEntry::INIT_ENTRY);
195    }
196
197    // Addon the log contents
198    out.extend_from_slice(new.as_str_ref().as_bytes());
199    // Add a newline
200    out.push(b'\n');
201
202    /*
203     * Make sure we are not stalled, because if we are we will interrupt the user shell with our message, and
204     * then check this message isn't from the init because if so it will have already been reported once
205     */
206    if (state!().is_ok() && !from_init && self.shout)
207    {
208      for line in (new.as_str_ref().split('\n'))
209      {
210        log(Status(format!("{} {}", format!("service@{}:", self.name).bold(), line)));
211      }
212    }
213
214    /*
215     * Reading from the logfile changes the cursor position so we set it back to the beginning to overwrite
216     * the file instead of appending it
217     */
218    self.file.rewind().into_trace(Error::ServiceLogContent)?;
219
220    // Compress the newly formed contents, saving to the log file
221    compress(Cursor::new(out), &mut self.file, CompressionLevel::Fastest);
222    self.line += 1;
223
224    Ok(())
225  }
226}

228impl<'entry> LogEntry<'entry>
229{
230  /// Marker added to entries to show the message is from the init system
231  pub const INIT_ENTRY: u8 = 0x8f;
232
233  /// Strip down to inner value, discarding init entry status
234  #[must_use]
235  pub fn value(&self) -> &'entry str
236  {
237    match (self)
238    {
239      Self::Service(val) | Self::Init(val) => val
240    }
241  }
