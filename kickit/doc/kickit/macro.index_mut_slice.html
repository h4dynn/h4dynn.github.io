macro_rules! index_mut_slice {
    {$array: ident [$begin: tt .. $end: tt] = [$($item: expr),+];} => { ... };
    {$array: ident [.. $end: tt] = $assign: expr;} => { ... };
    {$array: ident [$begin: tt .. $end: tt] = $assign: expr;} => { ... };
}
Expand description

Send bytes on RHS to a slice of an array (or vector) in iterated order

§Example

let mut buf = [0u8; 10];

index_mut_slice!{ buf[3..7] = [0xb0, 0x0b, 0x1e, 0x50]; }

// Indexed slice of the array was assigned!
assert_eq!(buf, [0, 0, 0, 0xb0, 0x0b, 0x1e, 0x50, 0, 0, 0]);
